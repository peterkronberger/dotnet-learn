== Database ==
CREATE TABLE "Incidents" (
    "Id" INTEGER NOT NULL CONSTRAINT "PK_Incidents" PRIMARY KEY AUTOINCREMENT,
    "User" TEXT NOT NULL,
    "Source" TEXT NOT NULL,
    "Status" TEXT NOT NULL,
    "FailureCount" INTEGER NOT NULL
);


CREATE TABLE "Events" (
    "Id" INTEGER NOT NULL CONSTRAINT "PK_Events" PRIMARY KEY AUTOINCREMENT,
    "Timestamp" TEXT NOT NULL,
    "EventType" TEXT NOT NULL,
    "User" TEXT NULL,
    "StoredIncidentId" INTEGER NOT NULL,
    CONSTRAINT "FK_Events_Incidents_StoredIncidentId" FOREIGN KEY ("StoredIncidentId") REFERENCES "Incidents" ("Id") ON DELETE CASCADE
);


CREATE INDEX "IX_Events_StoredIncidentId" ON "Events" ("StoredIncidentId");


CREATE INDEX "IX_Incidents_User" ON "Incidents" ("User");
